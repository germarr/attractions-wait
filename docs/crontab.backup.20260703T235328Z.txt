# YouTube V2 Cron Configuration — VM
# ============================================
#
# Install path on the VM: /home/azureuser/youtube_homelab
# (Differs from the current PC, which uses /home/gmarr/youtube/v2.)
#
# Install on the VM:
#   crontab /home/azureuser/youtube_homelab/transfer_to_vm/crontab.txt
#   crontab -l    # verify
#
# IMPORTANT: only one machine should run these jobs at a time.
# Before installing here, on the old PC run:
#   crontab -l > ~/crontab.backup.txt
#   crontab -r
#   sudo systemctl stop youtube-v2.service
# Otherwise: double API spend + possible SQLite WAL corruption.

SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin
PYTHONPATH=/home/azureuser/youtube_homelab

# ============================================
# DATA COLLECTION - Hourly (DISABLED — replaced by category trending)
# ============================================
# 5 */6 * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/collect_trending.py >> logs/collection.log 2>&1

# ============================================
# DAILY BACKUP - 3 AM
# ============================================
# Compressed backup with integrity verification, removes backups >30 days old
0 3 * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/backup.py --compress >> logs/backup.log 2>&1

# ============================================
# DAILY HEALTH CHECK - 6 AM
# ============================================
# Verifies database integrity and data freshness
# Exit code: 0=healthy, 1=warning, 2=critical
0 6 * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/health_check.py >> logs/health.log 2>&1

# ============================================
# WEEKLY MAINTENANCE - Sunday 4 AM
# ============================================
# VACUUM, ANALYZE, cleanup
0 4 * * 0 cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/maintenance.py --all >> logs/maintenance.log 2>&1

# ============================================
# DAILY SUBSCRIBER TRACKING - 5 AM
# ============================================
0 5 * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/track_subscribers.py >> logs/subscribers.log 2>&1

# ============================================
# WEEKLY METRICS REFRESH - Sunday 6 AM
# ============================================
# Refreshes view/like/comment counts for 50 most recent videos per channel
0 6 * * 0 cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/track_recent_videos.py --refresh >> logs/track_videos.log 2>&1

# ============================================
# WEEKLY ARCHIVE - Monday 1 AM
# ============================================
# Archives last week's data to Parquet, removes archived data from DB.
# Run before maintenance for optimal cleanup.
0 1 * * 1 cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/weekly_archive.py >> logs/archive.log 2>&1

# ============================================
# VIDEO DETECTION - 8 AM & 7 PM
# ============================================
# Detects new videos from analyzed channels
0 8  * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/track_recent_videos.py --detect >> logs/track_videos.log 2>&1
0 19 * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/track_recent_videos.py --detect >> logs/track_videos.log 2>&1

# ============================================
# DAILY SNAPSHOT COLLECTION - 9 PM
# ============================================
# Performance metrics for videos in their first 7 days
0 21 * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/track_recent_videos.py --snapshot >> logs/track_videos.log 2>&1

# ============================================
# DAILY TRENDING SNAPSHOT - 9:30 PM
# ============================================
# Performance snapshots for trending tracked videos (7-day tracking)
30 21 * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/track_recent_videos.py --trending-snapshot >> logs/track_videos.log 2>&1

# ============================================
# CATEGORY TRENDING - 10:05 AM & 10:05 PM
# ============================================
# Collects category-based trending videos into main tables with fetch_category_id
5 10,22 * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/collect_category_trending.py >> logs/category_trending.log 2>&1

# ============================================
# DAILY PG SYNC - 3:30 AM
# ============================================
# Mirrors SQLite database to PostgreSQL (uses PGHOST/PGUSER/... from .env)
30 3 * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/sync_to_postgres.py >> logs/pg_sync.log 2>&1

# ============================================
# HOURLY PRECISION TRACKING
# ============================================
# Check active channels for new video/short publications
0 * * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/precision_tracker.py --check-channels >> logs/precision_tracker.log 2>&1
# Hourly performance snapshots for tracked videos
5 * * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/precision_tracker.py --collect-snapshots >> logs/precision_tracker.log 2>&1

# ============================================
# AUTO-TRANSCRIBE - 8:30 AM & 7:30 PM
# ============================================
# Transcribes + summarizes new videos from auto-transcribe-enabled channels.
# Runs 30 min after detection. Requires Azure Whisper credentials in .env
# (or local GPU + openai-whisper if you kept that path).
30 8  * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/auto_transcribe.py >> logs/auto_transcribe.log 2>&1
30 19 * * * cd /home/azureuser/youtube_homelab && .venv/bin/python3 scripts/auto_transcribe.py >> logs/auto_transcribe.log 2>&1
